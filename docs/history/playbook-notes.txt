# Migrate to NetworkManager
sudo bash <<'EOF'
set -euo pipefail

apt install -y network-manager

# Ensure NetworkManager is running
systemctl enable --now NetworkManager

# Preserve the original network configuration
cp -n /etc/network/interfaces /etc/network/interfaces.orig

# Detect and migrate simple DHCP Ethernet interfaces
while read -r iface; do
    echo "Migrating $iface to NetworkManager..."

    if ! nmcli connection show "Wired-$iface" >/dev/null 2>&1; then
        nmcli connection add \
            type ethernet \
            ifname "$iface" \
            con-name "Wired-$iface" \
            connection.autoconnect yes \
            ipv4.method auto \
            ipv6.method auto
    fi

    sed -i \
        -e "/^allow-hotplug ${iface}$/d" \
        -e "/^auto ${iface}$/d" \
        -e "/^iface ${iface} inet dhcp$/d" \
        /etc/network/interfaces

done < <(awk '$1 == "iface" && $3 == "inet" && $4 == "dhcp" {print $2}' \
    /etc/network/interfaces)

EOF
sudo reboot

entry = '@import url("window-controls.css");'

for version in ("gtk-3.0", "gtk-4.0"):

    path.parent.mkdir(parents=True, exist_ok=True)

    if path.is_symlink():
        print(f"Skipping symbolic link: {path}")
        continue

    existing = path.read_text() if path.exists() else ""

    if entry not in existing:
        if path.exists():
            backup = path.with_name("gtk.css.before-desktopuser")
            if not backup.exists():
                backup.write_text(existing)

        path.write_text(entry + "\n" + existing)

    print(f"Configured: {path}")

# XCursor image chunk identifier
IMAGE_TYPE = 0xfffd0002


def verify_cursor(path):
    data = path.read_bytes()

    if len(data) < 16 or data[:4] != b"Xcur":
        raise ValueError(f"Invalid XCursor file: {path}")

    header, version, ntoc = struct.unpack_from(
        "<III", data, 4
    )

    if header < 16 or header + 12 * ntoc > len(data):
        raise ValueError(f"Invalid XCursor header: {path}")

    frames = Counter()

    for i in range(ntoc):
        kind, subtype, offset = struct.unpack_from(
            "<III", data, header + 12 * i
        )

        if kind != IMAGE_TYPE:
            continue

        if offset + 36 > len(data):
            raise ValueError(f"Truncated chunk: {path}")

        (
            chunk_header,
            chunk_type,
            nominal_size,
            chunk_version,
            width,
            height,
            xhot,
            yhot,
            delay,
        ) = struct.unpack_from("<9I", data, offset)

        if chunk_type != IMAGE_TYPE:
            raise ValueError(f"Invalid image chunk: {path}")

        if nominal_size != subtype:
            raise ValueError(f"Invalid nominal size: {path}")

        if not width or not height:
            raise ValueError(f"Invalid dimensions: {path}")

        if xhot >= width or yhot >= height:
            raise ValueError(f"Invalid hotspot: {path}")

        if chunk_header < 36:
            raise ValueError(f"Invalid chunk header: {path}")

        start = offset + chunk_header
        end = start + width * height * 4

        if end > len(data):
            raise ValueError(f"Truncated image: {path}")

        frames[nominal_size] += 1

    missing = sizes - set(frames)

    if missing:
        raise ValueError(
            f"{path.name}: missing sizes {sorted(missing)}"
        )

    return frames

        SOURCE_SIZE
    )

    if not ptr:
        raise RuntimeError(f"Cannot load cursor: {theme}")

    try:
        img = ptr.contents
        pixels = []

        # Convert premultiplied ARGB to straight RGBA
        for i in range(img.width * img.height):
            p = img.pixels[i]

            a = (p >> 24) & 255
            r = (p >> 16) & 255
            g = (p >> 8) & 255
            b = p & 255

            if a:
                r = min(255, round(r * 255 / a))
                g = min(255, round(g * 255 / a))
                b = min(255, round(b * 255 / a))
            else:
                r = g = b = 0

            pixels.append((r, g, b, a))

        artwork = Image.new(
            "RGBA", (img.width, img.height)
        )
        artwork.putdata(pixels)

    finally:
        lib.XcursorImageDestroy(ptr)

    # Remove transparent padding
    bounds = artwork.getchannel("A").getbbox()

    if bounds is None:
        raise ValueError(f"Empty cursor: {theme}")

    # High-quality proportional downsampling
    artwork.thumbnail(
        (ARTWORK_SIZE, ARTWORK_SIZE),
        Image.Resampling.LANCZOS
    )

    # Center on transparent canvas
    canvas = Image.new(
        "RGBA",
        (CANVAS_SIZE, CANVAS_SIZE),
        (0, 0, 0, 0)
    )

    y = (CANVAS_SIZE - artwork.height) // 2

        gnupg \
        debsig-verify

    # Create temporary working directory
    WORKDIR=$(mktemp -d)
    trap 'rm -rf -- "$WORKDIR"' EXIT

    BASE_URL="https://downloads.1password.com/linux"
    FINGERPRINT="3FEF9748469ADBE15DA7CA80AC2D62742012EA22"
    KEY_ID="AC2D62742012EA22"

    # Download official repository signing key
    curl --fail --location --retry 3 \
        "$BASE_URL/keys/1password.asc" \
        -o "$WORKDIR/1password.asc"

    # Verify signing key fingerprint
    ACTUAL_FINGERPRINT=$(
        gpg --batch --show-keys --with-colons \
            "$WORKDIR/1password.asc" \
            | awk -F: '$1 == "fpr" && !seen++ {print $10}'
    )

    if [ "$ACTUAL_FINGERPRINT" != "$FINGERPRINT" ]; then
        echo "ERROR: Unexpected 1Password signing key." >&2
        exit 1
    fi

    # Convert signing key to binary keyring format
    gpg --batch --yes --dearmor \
        --output "$WORKDIR/1password.gpg" \
        "$WORKDIR/1password.asc"

    # Install repository signing key
    sudo install -d -m 0755 /usr/share/keyrings

    sudo install -m 0644 \
        "$WORKDIR/1password.gpg" \
        /usr/share/keyrings/1password-archive-keyring.gpg

    # Configure official stable APT repository
    sudo tee /etc/apt/sources.list.d/1password.list \
        > /dev/null <<'EOF'
deb [arch=amd64 signed-by=/usr/share/keyrings/1password-archive-keyring.gpg] https://downloads.1password.com/linux/debian/amd64 stable main
EOF

    # Download official package signature verification policy
    curl --fail --location --retry 3 \
        "$BASE_URL/debian/debsig/1password.pol" \
        -o "$WORKDIR/1password.pol"

    # Install debsig verification policy
    sudo install -d -m 0755 \
        "/etc/debsig/policies/$KEY_ID"

    sudo install -m 0644 \
        "$WORKDIR/1password.pol" \
        "/etc/debsig/policies/$KEY_ID/1password.pol"

    # Install debsig verification keyring
    sudo install -d -m 0755 \
        "/usr/share/debsig/keyrings/$KEY_ID"

    sudo install -m 0644 \
        "$WORKDIR/1password.gpg" \
        "/usr/share/debsig/keyrings/$KEY_ID/debsig.gpg"

    # Update package indexes and install both applications
    sudo apt update

    sudo apt install -y \
        1password \
        1password-cli

    echo "1Password installation complete."
)

# Install supplemental user-provided fonts
(
    set -Eeuo pipefail

    ARCHIVE="$HOME/Downloads/Windows-Fonts.zip"
    INSTALL_DIR="/usr/local/share/fonts/windows-supplemental"

    test -f "$ARCHIVE"

    sudo apt install -y fontconfig python3 unzip

    WORKDIR=$(mktemp -d)
    trap 'rm -rf -- "$WORKDIR"' EXIT

    mkdir -p "$WORKDIR/source" "$WORKDIR/fonts"

    unzip -q "$ARCHIVE" -d "$WORKDIR/source"

    # Select supplemental fonts by actual font family metadata.
    # Exclude classic web fonts already provided by the Debian package.
    python3 - "$WORKDIR" <<'PY'
from pathlib import Path
import subprocess
import shutil
import sys

root = Path(sys.argv[1])

families = (
    "Segoe UI",
    "Tahoma",
    "Calibri",
    "Cambria",
    "Candara",
    "Consolas",
    "Constantia",
    "Corbel",
    "Aptos",
    "Cascadia Code",
    "Cascadia Mono",
    "Lucida Console",
    "Palatino Linotype",
    "Franklin Gothic Medium",
    "Bahnschrift",
    "Microsoft Sans Serif",
)

# Do not replace Debian's existing emoji and symbol fonts.
excluded = (
    "Segoe UI Emoji",
    "Segoe UI Symbol",
    "Segoe UI Historic",
    "Segoe UI Assets",
    "Segoe UI MDL2 Assets",
)

def matches(name, prefix):
    name = name.casefold()
    prefix = prefix.casefold()
    return name == prefix or name.startswith(prefix + " ")

count = 0

for path in sorted((root / "source").rglob("*")):

    if not path.is_file():
        continue

    if path.suffix.lower() not in (".ttf", ".otf", ".ttc", ".otc"):
        continue

    result = subprocess.run(
        ["fc-scan", "--format", "%{family}\n", str(path)],
        capture_output=True,
        text=True,
        check=False,
    )

    if result.returncode != 0:
        print(f"Skipping invalid font: {path.name}")
        continue

    names = {
        name.strip()
        for line in result.stdout.splitlines()
        for name in line.split(",")
        if name.strip()
    }

    if any(matches(name, item)
           for name in names for item in excluded):
        continue

    if not any(matches(name, item)
               for name in names for item in families):
        continue

    target = root / "fonts" / path.name

    if target.exists():
        if target.read_bytes() != path.read_bytes():
            raise RuntimeError(f"Conflicting font: {path.name}")
        continue

    shutil.copyfile(path, target)

    print(f"Selected: {path.name} â€” {', '.join(sorted(names))}")
    count += 1

if not count:
    raise SystemExit("No eligible supplemental fonts found.")

print(f"Selected {count} supplemental font files.")
PY

    # Install system-wide without modifying APT-managed fonts
    sudo install -d -m 0755 "$INSTALL_DIR"

    find "$WORKDIR/fonts" -maxdepth 1 -type f -print0 |
        while IFS= read -r -d '' FONT; do
            sudo install -m 0644 "$FONT" "$INSTALL_DIR/"
        done

    # Refresh fontconfig caches
    sudo fc-cache -f

    echo "Supplemental font installation complete."
)

    # Win+L: Lock computer
    gsettings set "$SCHEMA" looking-glass-keybinding "[]"

    python3 <<'PY'
import ast
import subprocess

schema = "org.cinnamon.desktop.keybindings"

def get(schema, key):
    value = subprocess.check_output(
        ["gsettings", "get", schema, key], text=True
    ).strip()
    if value.startswith("@as "):
        value = value[4:]
    return ast.literal_eval(value)

def set_value(schema, key, value):
    subprocess.run(
        ["gsettings", "set", schema, key, repr(value)],
        check=True
    )

def custom_schema(identifier):
    return (
        f"{schema}.custom-keybinding:"
        f"/org/cinnamon/desktop/keybindings/"
        f"custom-keybindings/{identifier}/"
    )

identifiers = get(schema, "custom-list")

# Reuse an existing System Monitor shortcut
identifier = next(
    (
        item for item in identifiers
        if item != "__dummy__"
        and get(custom_schema(item), "command") == "gnome-system-monitor"
    ),
    None
)

# Otherwise allocate the next available identifier
if identifier is None:
    index = 0
    while f"custom{index}" in identifiers:
        index += 1
    identifier = f"custom{index}"

shortcut = custom_schema(identifier)

set_value(shortcut, "name", "System Monitor")
set_value(shortcut, "command", "gnome-system-monitor")
set_value(shortcut, "binding", ["<Primary><Shift>Escape"])

if identifier not in identifiers:
    identifiers.append(identifier)

# Force Cinnamon to refresh its custom shortcuts
identifiers = [i for i in identifiers if i != "__dummy__"]
set_value(schema, "custom-list", identifiers + ["__dummy__"])
set_value(schema, "custom-list", identifiers)

print("System Monitor shortcut configured:", identifier)
PY
)
