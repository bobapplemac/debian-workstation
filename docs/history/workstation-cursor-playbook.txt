# XCursor image chunk identifier
IMAGE_TYPE = 0xfffd0002


def verify_cursor(path):
    data = path.read_bytes()

    if len(data) < 16 or data[:4] != b"Xcur":
        raise ValueError(f"Invalid XCursor file: {path}")

    header, version, ntoc = struct.unpack_from(
        "<III", data, 4
    )

    if header < 16 or header + 12 * ntoc > len(data):
        raise ValueError(f"Invalid XCursor header: {path}")

    frames = Counter()

    for i in range(ntoc):
        kind, subtype, offset = struct.unpack_from(
            "<III", data, header + 12 * i
        )

        if kind != IMAGE_TYPE:
            continue

        if offset + 36 > len(data):
            raise ValueError(f"Truncated chunk: {path}")

        (
            chunk_header,
            chunk_type,
            nominal_size,
            chunk_version,
            width,
            height,
            xhot,
            yhot,
            delay,
        ) = struct.unpack_from("<9I", data, offset)

        if chunk_type != IMAGE_TYPE:
            raise ValueError(f"Invalid image chunk: {path}")

        if nominal_size != subtype:
            raise ValueError(f"Invalid nominal size: {path}")

        if not width or not height:
            raise ValueError(f"Invalid dimensions: {path}")

        if xhot >= width or yhot >= height:
            raise ValueError(f"Invalid hotspot: {path}")

        if chunk_header < 36:
            raise ValueError(f"Invalid chunk header: {path}")

        start = offset + chunk_header
        end = start + width * height * 4

        if end > len(data):
            raise ValueError(f"Truncated image: {path}")

        frames[nominal_size] += 1

    missing = sizes - set(frames)

    if missing:
        raise ValueError(
            f"{path.name}: missing sizes {sorted(missing)}"
        )

    return frames

import ctypes as C
from pathlib import Path

class CursorImage(C.Structure):
    _fields_ = [
        ("version", C.c_uint32),
        ("size", C.c_uint32),
        ("width", C.c_uint32),
        ("height", C.c_uint32),
        ("xhot", C.c_uint32),
        ("yhot", C.c_uint32),
        ("delay", C.c_uint32),
        ("pixels", C.POINTER(C.c_uint32)),
    ]

lib = C.CDLL("libXcursor.so.1")

lib.XcursorLibraryLoadImage.argtypes = [
    C.c_char_p,
    C.c_char_p,
    C.c_int,
]

lib.XcursorLibraryLoadImage.restype = (
    C.POINTER(CursorImage)
)

lib.XcursorImageDestroy.argtypes = [
    C.POINTER(CursorImage)
]

lib.XcursorImageDestroy.restype = None

sizes = [
    24, 30, 32, 36, 40, 42, 48, 54,
    56, 60, 64, 72, 80, 96, 112, 128,
]

themes = [
    "Windows-Aero-Light",
    "Windows-Black",
]

failures = 0
checks = 0

for theme in themes:
    root = Path("/usr/share/icons") / theme / "cursors"

    if not root.is_dir():
        raise SystemExit(f"Theme not installed: {theme}")

    for path in sorted(root.iterdir()):
        # Cinnamon preview artwork is not an XCursor binary.
        if path.name == "thumbnail.png":
            continue

        for size in sizes:
            checks += 1

            ptr = lib.XcursorLibraryLoadImage(
                path.name.encode(),
                theme.encode(),
                size,
            )

            if not ptr:
                print(f"FAIL: {theme}/{path.name} @ {size}")
                failures += 1
                continue

            img = ptr.contents

            if (
                img.size != size
                or img.xhot >= img.width
                or img.yhot >= img.height
            ):
                print(
                    f"FAIL: {theme}/{path.name} @ {size}"
                    f" (returned {img.size})"
                )
                failures += 1

            lib.XcursorImageDestroy(ptr)

    print(f"Checked: {theme}")

print(f"Total checks: {checks}")
print(f"Failures: {failures}")

if failures or not checks:
    raise SystemExit(1)

# 4. Generate high-quality, centered cursor theme thumbnails
# Always regenerate from the 128px cursor, not from an existing thumbnail.
# This changes previews only, never cursor artwork or hotspots.
sudo python3 <<'PY'
import ctypes as C
from pathlib import Path
from PIL import Image

CANVAS_SIZE = 32
ARTWORK_SIZE = 28
SOURCE_SIZE = 128

class CursorImage(C.Structure):
    _fields_ = [
        ("version", C.c_uint32),
        ("size", C.c_uint32),
        ("width", C.c_uint32),
        ("height", C.c_uint32),
        ("xhot", C.c_uint32),
        ("yhot", C.c_uint32),
        ("delay", C.c_uint32),
        ("pixels", C.POINTER(C.c_uint32)),
    ]

lib = C.CDLL("libXcursor.so.1")

lib.XcursorFilenameLoadImage.argtypes = [
    C.c_char_p, C.c_int
]
lib.XcursorFilenameLoadImage.restype = C.POINTER(CursorImage)

lib.XcursorImageDestroy.argtypes = [
    C.POINTER(CursorImage)
]
lib.XcursorImageDestroy.restype = None

for theme in ("Windows-Aero-Light", "Windows-Black"):
    root = Path("/usr/share/icons") / theme / "cursors"

    # Load high-resolution original artwork
    ptr = lib.XcursorFilenameLoadImage(
        str(root / "left_ptr").encode(),
        SOURCE_SIZE
    )

    if not ptr:
        raise RuntimeError(f"Cannot load cursor: {theme}")

    try:
        img = ptr.contents
        pixels = []

        # Convert premultiplied ARGB to straight RGBA
        for i in range(img.width * img.height):
            p = img.pixels[i]

            a = (p >> 24) & 255
            r = (p >> 16) & 255
            g = (p >> 8) & 255
            b = p & 255

            if a:
                r = min(255, round(r * 255 / a))
                g = min(255, round(g * 255 / a))
                b = min(255, round(b * 255 / a))
            else:
                r = g = b = 0

            pixels.append((r, g, b, a))

        artwork = Image.new(
            "RGBA", (img.width, img.height)
        )
        artwork.putdata(pixels)

    finally:
        lib.XcursorImageDestroy(ptr)

    # Remove transparent padding
    bounds = artwork.getchannel("A").getbbox()

    if bounds is None:
        raise ValueError(f"Empty cursor: {theme}")

    artwork = artwork.crop(bounds)

    # High-quality proportional downsampling
    artwork.thumbnail(
        (ARTWORK_SIZE, ARTWORK_SIZE),
        Image.Resampling.LANCZOS
    )

    # Center on transparent canvas
    canvas = Image.new(
        "RGBA",
        (CANVAS_SIZE, CANVAS_SIZE),
        (0, 0, 0, 0)
    )

    x = (CANVAS_SIZE - artwork.width) // 2
    y = (CANVAS_SIZE - artwork.height) // 2

    canvas.alpha_composite(artwork, (x, y))

    output = root / "thumbnail.png"
    canvas.save(output)
    output.chmod(0o644)

    print(f"Generated: {output}")
PY
